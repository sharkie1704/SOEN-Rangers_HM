import { Role } from "../../domain/reg/Role";

export type SeedAccount =
  | {
      id: string;
      name: string;
      email: string;
      password: string;
      role: Role.CUSTOMER;
      deliveryLat: number;
      deliveryLng: number;
    }
  | {
      id: string;
      name: string;
      email: string;
      password: string;
      role: Role.COURIER | Role.OPERATOR;
    }
  | {
      id: string;
      name: string;
      email: string;
      password: string;
      role: Role.RESTAURANT;
      restaurantId: string;
    };

export interface SeedMenuItem {
  name: string;
  description: string;
  imageUrl: string;
  priceCents: number;
  available: boolean;
}

export interface SeedRestaurant {
  id: string;
  name: string;
  category: string;
  description: string;
  imageUrl: string;
  status: "OPEN" | "PAUSED" | "CLOSED";
  menuItems: SeedMenuItem[];
}

export interface SeedData {
  accounts: SeedAccount[];
  restaurants: SeedRestaurant[];
}

export type SeedAccountRecord = {
  id: string;
  name: string;
  email: string;
  passwordHash: string;
  role: Role;
  deliveryLat: number | null;
  deliveryLng: number | null;
  restaurantId: string | null;
};

4. src/application/seed/SeedController.ts
------------------------------------------
import type { PasswordHasher } from "../../domain/reg/ports/PasswordHasher";
import type { SeedData, SeedAccountRecord } from "./SeedData";
import type { SeedRepository } from "./SeedRepository";

export class SeedController {
  constructor(
    private readonly repository: SeedRepository,
    private readonly hasher: PasswordHasher,
  ) {}

  async seed(data: SeedData): Promise<void> {
    const accounts: SeedAccountRecord[] = await Promise.all(
      data.accounts.map(async (account) => {
        const common = {
          id: account.id,
          name: account.name,
          email: account.email,
          passwordHash: await this.hasher.hash(account.password),
          role: account.role,
          deliveryLat: null,
          deliveryLng: null,
          restaurantId: null,
        };

        if (account.role === "CUSTOMER") {
          return {
            ...common,
            deliveryLat: account.deliveryLat,
            deliveryLng: account.deliveryLng,
          };
        }

        if (account.role === "RESTAURANT") {
          return { ...common, restaurantId: account.restaurantId };
        }

        return common;
      }),
    );

    await this.repository.seed({ accounts, restaurants: data.restaurants });
  }
}

describe("SeedController", () => {
  it("hashes account passwords and sends accounts, restaurants, and menus to the repository", async () => {
    const saved: Parameters<SeedRepository["seed"]>[0][] = [];
    const repository: SeedRepository = {
      seed: async (data) => {
        saved.push(data);
      },
    };
    const hasher: PasswordHasher = {
      hash: async (password) => `hashed:${password}`,
      verify: async () => false,
    };

    await new SeedController(repository, hasher).seed(seedData);

    expect(saved).toHaveLength(1);
    expect(saved[0]).toEqual({
      accounts: [
        {
          id: "customer-id",
          name: "Seed Customer",
          email: "customer@example.com",
          passwordHash: "hashed:customer-password",
          role: Role.CUSTOMER,
          deliveryLat: 45.5,
          deliveryLng: -73.5,
          restaurantId: null,
        },
        {
          id: "staff-id",
          name: "Seed Staff",
          email: "staff@example.com",
          passwordHash: "hashed:staff-password",
          role: Role.RESTAURANT,
          deliveryLat: null,
          deliveryLng: null,
          restaurantId: "restaurant-id",
        },
      ],
      restaurants: seedData.restaurants,
    });
  });

  it("propagates hashing failures without persisting partial data", async () => {
    let repositoryCalled = false;
    const repository: SeedRepository = {
      seed: async () => {
        repositoryCalled = true;
      },
    };
    const hasher: PasswordHasher = {
      hash: async () => {
        throw new Error("Hashing failed");
      },
      verify: async () => false,
    };

    await expect(new SeedController(repository, hasher).seed(seedData)).rejects.toThrow(
      "Hashing failed",
    );
    expect(repositoryCalled).toBe(false);
  });
});

6. src/technical-services/seed/loadSeedConfig.ts
-------------------------------------------------
import { readFile } from "node:fs/promises";
import { z } from "zod";
import { Role } from "../../domain/reg/Role";
import type { SeedData } from "../../application/seed/SeedData";

const nonEmptyString = z.string().trim().min(1);
const accountBase = {
  id: nonEmptyString,
  name: nonEmptyString,
  email: z.email(),
  password: z.string().min(1).max(72),
};

const accountSchema = z.discriminatedUnion("role", [
  z.object({
    ...accountBase,
    role: z.literal(Role.CUSTOMER),
    deliveryLat: z.number().min(-90).max(90),
    deliveryLng: z.number().min(-180).max(180),
  }),
  z.object({ ...accountBase, role: z.literal(Role.COURIER) }),
  z.object({
    ...accountBase,
    role: z.literal(Role.RESTAURANT),
    restaurantId: nonEmptyString,
  }),
  z.object({ ...accountBase, role: z.literal(Role.OPERATOR) }),
]);

const seedConfigSchema = z.object({
  accounts: z.array(accountSchema),
  restaurants: z.array(
    z.object({
      id: nonEmptyString,
      name: nonEmptyString,
      category: nonEmptyString,
      description: z.string(),
      imageUrl: z.string(),
      status: z.enum(["OPEN", "PAUSED", "CLOSED"]),
      menuItems: z.array(
        z.object({
          name: nonEmptyString,
          description: z.string(),
          imageUrl: z.string(),
          priceCents: z.number().int().nonnegative(),
          available: z.boolean(),
        }),
      ),
    }),
  ),
}).superRefine((config, context) => {
  const restaurantIds = new Set(config.restaurants.map((restaurant) => restaurant.id));
  const emails = new Set<string>();

  for (const [index, account] of config.accounts.entries()) {
    const email = account.email.trim().toLowerCase();
    if (emails.has(email)) {
      context.addIssue({
        code: "custom",
        path: ["accounts", index, "email"],
        message: "Seeded account emails must be unique",
      });
    }
    emails.add(email);

    if (account.role === Role.RESTAURANT && !restaurantIds.has(account.restaurantId)) {
      context.addIssue({
        code: "custom",
        path: ["accounts", index, "restaurantId"],
        message: `Unknown seeded restaurant "${account.restaurantId}"`,
      });
    }
  }
});

export async function loadSeedConfig(): Promise<SeedData> {
  const contents = await readFile(new URL("../../../config.json", import.meta.url), "utf8");
  const parsed = seedConfigSchema.safeParse(JSON.parse(contents));

  if (!parsed.success) {
    throw new Error(`Invalid seed configuration: ${parsed.error.message}`);
  }

  return parsed.data;
}


7. src/technical-services/seed/PrismaSeedRepository.ts
-------------------------------------------------------
import { prisma } from "../db/prismaClient";
import type { SeedAccountRecord, SeedRestaurant } from "../../application/seed/SeedData";
import type { SeedRepository } from "../../application/seed/SeedRepository";

export class PrismaSeedRepository implements SeedRepository {
  async seed(data: { accounts: SeedAccountRecord[]; restaurants: SeedRestaurant[] }): Promise<void> {
    await prisma.$transaction(async (transaction) => {
      for (const account of data.accounts) {
        const accountData = {
          name: account.name,
          passwordHash: account.passwordHash,
          role: account.role,
          deliveryLat: account.deliveryLat,
          deliveryLng: account.deliveryLng,
          restaurantId: account.restaurantId,
        };

        await transaction.user.upsert({
          where: { email: account.email },
          create: {
            id: account.id,
            email: account.email,
            ...accountData,
          },
          update: accountData,
        });
      }

      for (const restaurant of data.restaurants) {
        const { menuItems, ...restaurantData } = restaurant;
        await transaction.restaurant.upsert({
          where: { id: restaurant.id },
          create: restaurantData,
          update: restaurantData,
        });

        for (const menuItem of menuItems) {
          const existing = await transaction.menuItem.findFirst({
            where: { restaurantId: restaurant.id, name: menuItem.name },
            select: { id: true },
          });

          if (existing) {
            await transaction.menuItem.update({
              where: { id: existing.id },
              data: menuItem,
            });
          } else {
            await transaction.menuItem.create({
              data: { ...menuItem, restaurantId: restaurant.id },
            });
          }
        }
      }
    });
  }
}

import { SeedController } from "./application/seed/SeedController";
import { loadSeedConfig } from "./technical-services/seed/loadSeedConfig";
import { PrismaSeedRepository } from "./technical-services/seed/PrismaSeedRepository";

async function startServer(): Promise<void> {
  if (process.env.SEED_FROM_CONFIG === "true") {
